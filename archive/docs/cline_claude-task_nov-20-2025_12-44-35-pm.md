                        {
                            response.getJsonArray(APIConstants.Entity.OBJECT.getName()).add(item);
                        }
                    }
                }
            }

            var items = IntegrationConfigStore.getStore().getItemsByMapValueField(Integration.INTEGRATION_CONTEXT, IntegrationProfile.INTEGRATION_CREDENTIAL_PROFILE, id);

            for (var index = 0; index < items.size(); index++)
            {
                var item = items.getJsonObject(index);

                item.remove(Integration.INTEGRATION_ATTRIBUTES);

                addEntity(response, APIConstants.Entity.INTEGRATION.getName());

                response.getJsonArray(APIConstants.Entity.INTEGRATION.getName()).add(item);
            }

                    ConfigConstants.removeGarbageFields(object);

                    item.mergeIn(object);
                }
            }
        }
        catch (Exception exception)

            LOGGER.error(exception);
        }
        return Future.succeededFuture(response);
    }

    private void prepareContext(JsonObject requestBody, Promise<JsonObject> promise)
    {
        try
        {
            var context = requestBody.getJsonObject(CREDENTIAL_PROFILE_CONTEXT);

            if (context.containsKey(OAuthUtil.AUTHENTICATION_TYPE) && NMSConstants.AuthenticationType.OAUTH.getName().equalsIgnoreCase(context.getString(OAuthUtil.AUTHENTICATION_TYPE)) && context.containsKey(SESSION_ID) && NMSConstants.OAuthGrantType.AUTHORIZATION_CODE.getName().equalsIgnoreCase(context.getString(OAuthUtil.GRANT_TYPE)))
            {
                Bootstrap.vertx().eventBus().<JsonObject>request(EventBusConstants.EVENT_OAUTH_CONTEXT_FETCH, context,
                        new DeliveryOptions().setSendTimeout(2 * 1000L),
                        reply ->
                        {
                            if (reply.succeeded())
                            {
                                context.mergeIn(reply.result().body());

                                context.remove(SESSION_ID);

                                promise.complete(requestBody);
                            }
                            else
                            {
                                promise.fail(reply.cause());
                            }
                        });
            }
            else if (context.containsKey(OAuthUtil.AUTHENTICATION_TYPE) && AuthenticationType.CERTIFICATE.getName().equalsIgnoreCase(context.getString(OAuthUtil.AUTHENTICATION_TYPE)) && context.containsKey(CLIENT_CERTIFICATE_CONFIGURATION))
            {
                if (context.getString(CLIENT_CERTIFICATE_CONFIGURATION).equalsIgnoreCase(ConfigurationType.UPLOAD.getName()))
                {
                    if (context.containsKey(CLIENT_CERTIFICATE_FILE) && context.containsKey(CLIENT_KEY_FILE) && context.containsKey(CERTIFICATE_AUTHORITY_FILE))
                    {
                        context.put(CLIENT_KEY, new String(Base64.decodeBase64(Bootstrap.vertx().fileSystem().readFileBlocking(GlobalConstants.CURRENT_DIR + GlobalConstants.PATH_SEPARATOR + GlobalConstants.UPLOADS + GlobalConstants.PATH_SEPARATOR + context.getJsonObject(CLIENT_KEY_FILE).getString(RESULT)).toString())));

                        context.put(CLIENT_CERTIFICATE, new String(Base64.decodeBase64(Bootstrap.vertx().fileSystem().readFileBlocking(GlobalConstants.CURRENT_DIR + GlobalConstants.PATH_SEPARATOR + GlobalConstants.UPLOADS + GlobalConstants.PATH_SEPARATOR + context.getJsonObject(CLIENT_CERTIFICATE_FILE).getString(RESULT)).toString())));

                        context.put(CERTIFICATE_AUTHORITY, new String(Base64.decodeBase64(Bootstrap.vertx().fileSystem().readFileBlocking(GlobalConstants.CURRENT_DIR + GlobalConstants.PATH_SEPARATOR + GlobalConstants.UPLOADS + GlobalConstants.PATH_SEPARATOR + context.getJsonObject(CERTIFICATE_AUTHORITY_FILE).getString(RESULT)).toString())));

                        promise.complete(requestBody);
                    }
                    else
                    {
                        promise.fail("Missing required certificate files");
                    }
                }
                else if (context.getString(CLIENT_CERTIFICATE_CONFIGURATION).equalsIgnoreCase(ConfigurationType.MANUAL.getName()))
                {
                    if (context.containsKey(CLIENT_CERTIFICATE) && context.containsKey(CLIENT_KEY) && context.containsKey(CERTIFICATE_AUTHORITY))
                    {
                        context.put(CLIENT_KEY, new String(Base64.decodeBase64(context.getString(CLIENT_KEY))));

                        context.put(CLIENT_CERTIFICATE, new String(Base64.decodeBase64(context.getString(CLIENT_CERTIFICATE))));

                        context.put(CERTIFICATE_AUTHORITY, new String(Base64.decodeBase64(context.getString(CERTIFICATE_AUTHORITY))));

                        promise.complete(requestBody);
                    }
                    else
                    {
                        promise.fail("Missing required certificate fields");
                    }
                }
                else
                {
                    promise.fail("Invalid certificate configuration");
                }
            }
            else
            {
                promise.complete(requestBody);
            }
        }
        catch (Exception exception)
        {
            LOGGER.error(exception);

            promise.fail(exception);
        }
    }
